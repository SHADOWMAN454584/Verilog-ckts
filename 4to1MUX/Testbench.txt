module tb_4to1mux();
reg d0,d1,d2,d3,s0,s1;

wire y;

mux4to1 uut(
    .d0(d0),
    .d1(d1),
    .d2(d2),
    .d3(d3),
    .s0(s0),
    .s1(s1),
    .y(y)
);

always begin    
    #10
    s1=0;s0=0;d0=1;d1=0;d2=0;d3=0;
    #10
    s1=0;s0=1;d0=0;d1=1;d2=0;d3=0;
    #10
    s1=1;s0=0;d0=0;d1=0;d2=1;d3=0;
    #10
    s1=1;s0=1;d0=0;d1=0;d2=0;d3=1;
    #10
    
    $finish;
    end 
endmodule
