module tb_comparator_2bit();

    reg [1:0]A;
    reg [1:0]B;
    
    wire x,y,z;
    integer i,j;
    
    comparator_2bit dut (
        .A(A),
        .B(B),
        .y(y),
        .x(x),
        .z(z)
    );
    
    initial begin
        for (i = 0;i < 4 ; i = i + 1) begin
            for (j = 0;j < 4 ; j = j + 1) begin
                A = i;
                B = j;
                #10;
            end
        end
    end
endmodule
