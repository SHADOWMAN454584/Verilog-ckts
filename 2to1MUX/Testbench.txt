module tb_MUX_2to1();

    reg  i0, i1, s, e;
    wire y;

    // Instantiate DUT
    MUX_2to1 uut (
        .i0(i0), .i1(i1), .s(s), .e(e), .y(y)
    );

    initial begin
        // Only drive the input signals over time
        e = 0; s = 0; i0 = 1; i1 = 0; #10;
        e = 0; s = 1; #10;
        e = 1; s = 0; #10;
        e = 1; s = 1; #10;
    end

endmodule
