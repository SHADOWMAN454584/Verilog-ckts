module tb_alu_2bit();
    reg [1:0] A;
    reg [1:0] B;
    reg [1:0] sel;
    wire [1:0]alu_out;
    wire carry; 
    
    alu_2bit dut (
        .A(A),
        .B(B),
        .sel(sel),
        .alu_out(alu_out),
        .carry(carry)
    );
    
    initial begin
        A = 2'b10; B = 2'b01; sel = 2'b00; #10;
        A = 2'b11; B = 2'b10; sel = 2'b00; #10;
        A = 2'b11; B = 2'b01; sel = 2'b01; #10;
        A = 2'b01; B = 2'b10; sel = 2'b01; #10;
        A = 2'b11; B = 2'b10; sel = 2'b10; #10;     
        A = 2'b01; B = 2'b10; sel = 2'b11; #10;
    end
endmodule
